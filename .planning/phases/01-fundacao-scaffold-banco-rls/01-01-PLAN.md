```sql
-- 0001_schema.sql — 4 tabelas, constraints, índices, trigger de profile
-- Nomes de entidade em inglês (CLAUDE.md): Grupo->group, Participante->participant, Sorteio->draw

create extension if not exists "pgcrypto";

-- profile: espelha auth.users. id É o auth.users.id (não uma PK própria).
create table public.profile (
  id            uuid primary key references auth.users (id) on delete cascade,
  email         text not null,
  display_name  text not null,
  avatar_url    text,
  created_at    timestamptz not null default now()
);

create table public.group (
  id                uuid primary key default gen_random_uuid(),
  name              text not null check (char_length(name) between 3 and 60),
  admin_id          uuid not null references public.profile (id) on delete cascade,
  status            text not null default 'open'
                      check (status in ('open', 'drawing', 'closed')),
  invite_token      text not null unique default encode(gen_random_bytes(16), 'hex'),
  min_participants  int  not null default 3 check (min_participants >= 3),
  created_at        timestamptz not null default now()
);

create table public.participant (
  id            uuid primary key default gen_random_uuid(),
  group_id      uuid not null references public.group (id) on delete cascade,
  user_id       uuid not null references public.profile (id) on delete cascade,
  display_name  text not null,
  wish_list     text,
  joined_at     timestamptz not null default now()
);

create table public.draw (
  id           uuid primary key default gen_random_uuid(),
  group_id     uuid not null references public.group (id) on delete cascade,
  giver_id     uuid not null references public.participant (id) on delete cascade,
  receiver_id  uuid not null references public.participant (id) on delete cascade,
  created_at   timestamptz not null default now()
);

-- === Constraints (ARCHITECTURE.md Problema 3) ===

-- DB-02: um usuário não pode ter duas linhas de participante no mesmo grupo
alter table public.participant
  add constraint participant_group_user_unique unique (group_id, user_id);

-- DB-04: ninguém tira a si mesmo (defesa em profundidade sobre a garantia do algoritmo)
alter table public.draw
  add constraint draw_no_self_gift check (giver_id <> receiver_id);

-- DB-03: perfect matching — cada participante dá exatamente uma vez...
alter table public.draw
  add constraint draw_giver_unique unique (group_id, giver_id);

-- DB-03: ...e recebe exatamente uma vez
alter table public.draw
  add constraint draw_receiver_unique unique (group_id, receiver_id);

-- === Índices usados pelas policies (DB-09) — Postgres não indexa FK automaticamente ===
create index idx_participant_group_id on public.participant (group_id);
create index idx_participant_user_id  on public.participant (user_id);
create index idx_draw_group_id        on public.draw (group_id);
create index idx_draw_giver_id        on public.draw (giver_id);
create index idx_group_admin_id       on public.group (admin_id);

-- === Trigger handle_new_user: popula profile ao criar conta ===
-- Consumido pela AUTH-06 na Fase 2, mas é infraestrutura de banco (DB-01), criado aqui.
create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
begin
  insert into public.profile (id, email, display_name)
  values (
    new.id,
    new.email,
    coalesce(new.raw_user_meta_data ->> 'display_name', split_part(new.email, '@', 1))
  );
  return new;
end;
$$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function public.handle_new_user();
```

```sql
-- 0002_private_helpers.sql — DB-05
-- Essas funções são a base de TODAS as policies. Precisam existir antes de qualquer
-- CREATE POLICY. Ficam em schema `private` (nunca `public`) porque qualquer função em
-- schema exposto pela Data API é chamável diretamente por qualquer cliente autenticado.
-- `set search_path = ''` + nomes schema-qualificados é obrigatório em SECURITY DEFINER.

create schema if not exists private;

create or replace function private.my_participant_ids()
returns setof uuid
language sql
security definer
set search_path = ''
stable
as $$
  select id from public.participant
  where user_id = (select auth.uid())
$$;

create or replace function private.my_group_ids()
returns setof uuid
language sql
security definer
set search_path = ''
stable
as $$
  select group_id from public.participant
  where user_id = (select auth.uid())
$$;

revoke execute on function private.my_participant_ids() from public;
revoke execute on function private.my_group_ids() from public;
grant usage on schema private to authenticated;
grant execute on function private.my_participant_ids() to authenticated;
grant execute on function private.my_group_ids() to authenticated;
```

```sql
-- 0003_rls_policies.sql — DB-06, DB-07, DB-08
-- Depende de 0002_private_helpers.sql (as funções private.* precisam existir antes).

-- === DB-06: habilitar RLS nas 4 tabelas ===
alter table public.profile     enable row level security;
alter table public.group       enable row level security;
alter table public.participant enable row level security;
alter table public.draw        enable row level security;

-- === profile ===
create policy "profile: ver o proprio perfil" on public.profile
for select to authenticated
using ( id = (select auth.uid()) );

create policy "profile: editar o proprio perfil" on public.profile
for update to authenticated
using ( id = (select auth.uid()) )
with check ( id = (select auth.uid()) );
-- INSERT em profile é feito pelo trigger handle_new_user (SECURITY DEFINER),
-- nunca pelo cliente — por isso não existe policy de INSERT aqui.

-- === group ===
create policy "group: admin le seus grupos" on public.group
for select to authenticated
using ( admin_id = (select auth.uid()) );

create policy "group: participante le grupos onde esta" on public.group
for select to authenticated
using ( id in (select private.my_group_ids()) );

create policy "group: admin cria grupo" on public.group
for insert to authenticated
with check ( admin_id = (select auth.uid()) );

create policy "group: admin atualiza seu grupo" on public.group
for update to authenticated
using ( admin_id = (select auth.uid()) )
with check ( admin_id = (select auth.uid()) );

-- === participant ===
create policy "participant: ver colegas do grupo" on public.participant
for select to authenticated
using ( group_id in (select private.my_group_ids()) );

create policy "participant: editar os proprios dados" on public.participant
for update to authenticated
using ( user_id = (select auth.uid()) )
with check ( user_id = (select auth.uid()) );

create policy "participant: entrar em um grupo" on public.participant
for insert to authenticated
with check (
  user_id = (select auth.uid())
  and group_id in (select g.id from public.group g where g.status = 'open')
);

create policy "participant: admin remove participante" on public.participant
for delete to authenticated
using (
  group_id in (select g.id from public.group g where g.admin_id = (select auth.uid()))
);

-- === draw — A POLÍTICA CRÍTICA DO PROJETO (DB-07, DB-08) ===
-- ÚNICA leitura permitida: a própria linha, como giver. SEM exceção para admin.
-- Se alguém algum dia adicionar `or ... admin_id ...` aqui, o core value do projeto morre.
create policy "draw: ver apenas meu proprio par" on public.draw
for select to authenticated
using ( giver_id in (select private.my_participant_ids()) );

-- DB-08: NENHUMA policy de INSERT/UPDATE/DELETE em draw para `authenticated`.
-- A ausência é deliberada e é o mecanismo: a única escrita vem da Edge Function
-- autenticada com a service_role/secret key, que bypassa RLS por design.
-- NÃO adicionar policy de escrita aqui em nenhuma fase futura.
```
