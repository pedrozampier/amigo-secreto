```typescript
import type { HttpErrorResponse, HttpInterceptorFn } from '@angular/common/http';
import { catchError, throwError } from 'rxjs';

/**
 * Interceptor funcional de erro centralizado (RA6-ID22).
 *
 * ESCOPO REAL, documentado deliberadamente (D-15): este interceptor cobre
 * EXCLUSIVAMENTE o tráfego do `HttpClient` do Angular — na prática, a chamada à
 * BrasilAPI. O `supabase-js` usa o `fetch` nativo próprio e NUNCA passa pelo
 * `HttpClient`, então nenhum erro de Supabase chega aqui. Isso não é uma limitação
 * a corrigir: é uma consequência da arquitetura (ARCHITECTURE.md, "o conflito do
 * ID22"), e o cliente Supabase já anexa o JWT sozinho — não há token a injetar.
 *
 * Responsabilidade: registrar a falha num ponto único e normalizá-la em
 * `HttpFalha`, de forma que cada chamador não precise reinterpretar status HTTP.
 * Re-lança o erro (não engole) para que o `catchError` do HolidayService ainda
 * decida o estado de UI — D-14: falhar na API não pode quebrar o dashboard.
 */
export interface HttpFalha {
  status: number;
  mensagem: string;
}

export const errorInterceptor: HttpInterceptorFn = (req, next) =>
  next(req).pipe(
    catchError((erro: HttpErrorResponse) => {
      const falha: HttpFalha = {
        status: erro.status,
        mensagem:
          erro.status === 0
            ? 'Sem conexão com a internet.'
            : `A requisição falhou (HTTP ${erro.status}).`,
      };
      // Ponto único de observabilidade de falhas HTTP do projeto.
      console.error('[HTTP]', req.method, req.url, falha.status, falha.mensagem);
      return throwError(() => falha);
    }),
  );
```

**2a. Criar `src/app/core/services/holiday.service.ts`:**

```typescript
import { Injectable, computed, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { toSignal } from '@angular/core/rxjs-interop';
import { catchError, of } from 'rxjs';
import { diasAteNatal } from '../dias-ate-natal';

export interface Holiday {
  /** 'YYYY-MM-DD' */
  date: string;
  name: string;
  type: string;
}

@Injectable({ providedIn: 'root' })
export class HolidayService {
  private readonly http = inject(HttpClient);
  private readonly ano = new Date().getFullYear();

  /**
   * RA6-ID19 + RA6-ID24: GET real a uma API pública, atravessando a ponte
   * RxJS→Signals com `toSignal()`.
   *
   * NÃO trocar por `httpResource()` (que existe no Angular 21): o ID24 grada
   * especificamente a ponte Observable→Signal, e `httpResource` não a demonstra.
   *
   * Três estados distintos sem signal de loading separado:
   *   `undefined` = carregando · `null` = falhou · array = carregado.
   * O `catchError` → `of(null)` é o que cumpre D-14: a falha vira estado de UI em
   * vez de quebrar o dashboard. O interceptor já logou e normalizou a falha antes.
   */
  private readonly feriados = toSignal(
    this.http
      .get<Holiday[]>(`https://brasilapi.com.br/api/feriados/v1/${this.ano}`)
      .pipe(catchError(() => of(null))),
    { initialValue: undefined },
  );

  /** O feriado de Natal da resposta da API — fonte da data exibida com DatePipe (ID7). */
  readonly natal = computed<Holiday | null | undefined>(() => {
    const lista = this.feriados();
    if (lista === undefined) return undefined;
    if (lista === null) return null;
    return lista.find((f) => f.date === `${this.ano}-12-25`) ?? null;
  });

  /**
   * API-02. `undefined` enquanto carrega, `null` se a API falhou (ou não trouxe o
   * Natal), número quando pronto.
   *
   * O `ano` vem do campo `date` DA RESPOSTA, não de `new Date()` local — é o que
   * mantém o GET à API pública não-decorativo (D-14): sem a resposta, não há número.
   */
  readonly diasAteONatal = computed<number | null | undefined>(() => {
    const feriado = this.natal();
    if (feriado === undefined) return undefined;
    if (feriado === null) return null;
    const anoDaResposta = Number(feriado.date.slice(0, 4));
    return diasAteNatal(new Date(), anoDaResposta);
  });
}
```
