/** GRP-09. Uma query só: as duas policies de SELECT em `group` são ORed pela RLS,
 *  então `select('*')` sem filtro devolve a união. A separação é no cliente. */
async refreshDashboard(): Promise<{ error: string | null }> {
  const uid = this.auth.user()?.id ?? null;
  const { data, error } = await this.supabase
    .from('group')
    .select('*')
    .order('created_at', { ascending: false });
  if (error) { return { error: 'Não foi possível carregar seus grupos.' }; }
  const grupos = (data ?? []) as Group[];
  this._administered.set(grupos.filter((g) => g.admin_id === uid));

**DECISÃO DE PLANEJAMENTO — desvio explícito e rastreável de D-08.**

D-08 diz literalmente: "`groupResolver` busca o grupo uma vez; `adminGuard` lê esse
resultado". Isso **não é mecanicamente possível** no Angular Router: na MESMA rota, os
`canActivate` rodam ANTES dos `resolve`, então um guard nunca consegue ler o `route.data`
produzido por um resolver irmão. As três saídas eram: (a) embutir a checagem de admin no
próprio resolver, devolvendo `UrlTree`; (b) guard e resolver com queries independentes
(= duas roundtrips, exatamente o que D-08 rejeitou); (c) guard na rota pai e resolver na
filha (ainda duas queries).

**Escolhida: (a).** Um `ResolveFn` pode devolver `UrlTree` para cancelar-e-redirecionar —
comportamento oficial e documentado do Router, o mesmo mecanismo que o `authGuard` já usa.
Isso preserva o OBJETIVO declarado de D-08 ("uma query por navegação") mesmo abandonando
sua letra. Não criar `src/app/core/guards/admin.guard.ts`: um guard que refizesse a query
seria a opção (b) rejeitada, e um que não a refizesse leria `undefined`.

Cobertura do RA5-ID18 ("Functional Guards + Resolvers") fica intacta: o `authGuard`
funcional da Fase 2 protege a rota pai de layout, e o `groupResolver` funcional é o resolver.
**Registrar este desvio no SUMMARY desta plan** para que fique rastreável na banca.

```typescript
import { inject } from '@angular/core';
import { Router, type ResolveFn } from '@angular/router';
import type { Group } from '../models/group.model';
import { AuthService } from '../services/auth.service';
import { GroupService } from '../services/group.service';

/**
 * Resolver funcional de /group/:id/manage (RA5-ID18).
 *
 * DESVIO DOCUMENTADO DE D-08: a decisão original previa um `adminGuard` separado
 * lendo o resultado deste resolver. O Angular Router roda os `canActivate` ANTES
 * dos `resolve` da MESMA rota, então o guard leria `undefined`. A checagem de admin
 * vive aqui, devolvendo um `UrlTree` — retorno oficialmente suportado por `ResolveFn`
 * — o que preserva o objetivo real de D-08: UMA query por navegação.
 *
 * Também cobre o caso "grupo não existe / a RLS não me deixa vê-lo": ambos chegam
 * como `group: null` e levam ao mesmo redirect, sem distinguir os dois para o
 * usuário (não vazar a existência de um grupo de outra pessoa).
 */
export const groupResolver: ResolveFn<Group | import('@angular/router').UrlTree> = async (route) => {
  const groups = inject(GroupService);
  const auth = inject(AuthService);
  const router = inject(Router);

  const id = route.paramMap.get('id');
  if (!id) {
    return router.createUrlTree(['/dashboard']);
  }

  const { group } = await groups.loadGroup(id);
  if (!group || group.admin_id !== auth.user()?.id) {
    return router.createUrlTree(['/dashboard']);
  }

  return group;
};
```

inline, para o ESLint ficar contente. A assinatura final deve ser
`ResolveFn<Group | UrlTree>`.
