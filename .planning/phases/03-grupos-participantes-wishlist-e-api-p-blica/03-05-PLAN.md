**1a. Criar `src/app/features/group/manage/participant-row.ts` + `.html`** — existe para
entregar o RA4-ID13 (`input()` + `output()`) com um caso de uso real:

```typescript
import { Component, input, output } from '@angular/core';
import { HlmButton } from '@spartan-ng/helm/button';
import type { Participant } from '../../../core/models/group.model';

/**
 * Item da lista de participantes (RA4-ID13): recebe dados por `input()` e emite o
 * pedido de remoção por `output()` — funções, nunca os decorators @Input/@Output
 * (o ESLint quebra o build neles).
 *
 * Confirmação em dois passos INLINE, controlada pelo input `confirmando`: o pai
 * decide qual linha está em confirmação. Nenhum dialog — a primitiva Spartan
 * `dialog` puxaria @ng-icons, ausente do SDD §2.
 */
@Component({
  selector: 'app-participant-row',
  imports: [HlmButton],
  templateUrl: './participant-row.html',
})
export class ParticipantRow {
  readonly participant = input.required<Participant>();
  /** true quando o pai está aguardando a confirmação de remoção DESTA linha. */
  readonly confirmando = input(false);
  /** Mostra os controles de remoção apenas para o admin do grupo. */
  readonly podeRemover = input(false);

  readonly pedirRemocao = output<string>();
  readonly confirmarRemocao = output<string>();
  readonly cancelarRemocao = output<void>();
}
```

`participant-row.html` — nome à esquerda, controles à direita, legível em 320px:
```html
<div class="flex items-center justify-between gap-2 border-b py-3 last:border-b-0">
  <span class="min-w-0 flex-1 truncate text-sm text-foreground">
    {{ participant().display_name }}
  </span>

  @if (podeRemover()) {
    @if (confirmando()) {
      <div class="flex shrink-0 items-center gap-1">
        <span class="text-xs text-muted-foreground">Remover?</span>
        <button hlmBtn variant="destructive" size="sm" type="button"
                (click)="confirmarRemocao.emit(participant().id)">Sim</button>
        <button hlmBtn variant="outline" size="sm" type="button"
                (click)="cancelarRemocao.emit()">Não</button>
      </div>
    } @else {
      <button hlmBtn variant="outline" size="sm" type="button" class="shrink-0"
              [attr.aria-label]="'Remover ' + participant().display_name"
              (click)="pedirRemocao.emit(participant().id)">Remover</button>
    }
  }
</div>
```

    const { preview, error } = await this.groups.previewByToken(this.token());
